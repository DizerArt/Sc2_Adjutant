def col(scene,name):
    c=bpy.data.collections.new(name)
    scene.collection.children.link(c)
    return c

upper=col(library,'Буквы — A–Z и подчёркивание')
lower=col(library,'Буквы — a–z')
support=col(library,'Буквы — просмотр библиотеки')

def camera(scene,name,width,xy,collection):
    data=bpy.data.cameras.new(name)
    data.type=next(e.identifier for e in data.bl_rna.properties['type'].enum_items if e.identifier=='ORTHO')
    data.ortho_scale=width
    data.clip_start=.01
    data.clip_end=100
    obj=bpy.data.objects.new(name,data)
    obj.location=(xy[0],xy[1],30)
    collection.objects.link(obj)
    scene.camera=obj
    return obj

camera(library,'Буквы — камера образца',34,(0,.1),support)

def flat(name,color):
    mat=bpy.data.materials.new(name)
    mat.use_nodes=True
    nodes=mat.node_tree.nodes
    nodes.clear()
    em=nodes.new('ShaderNodeEmission')
    em.inputs[0].default_value=color
    output=nodes.new('ShaderNodeOutputMaterial')
    mat.node_tree.links.new(em.outputs[0],output.inputs['Surface'])
    return mat

def quad(name,coords,collection,mat):
    mesh=bpy.data.meshes.new(name+' — UV плоскость')
    mesh.from_pydata(coords,[],[(0,1,2,3)])
    obj=bpy.data.objects.new(name,mesh)
    collection.objects.link(obj)
    mesh.materials.append(mat)
    return obj

back=flat('Буквы — графитовый фон',(.018,.023,.03,1))
quad('Буквы — нейтральный задник',[(-20,-14,-1),(20,-14,-1),(20,14,-1),(-20,14,-1)],support,back)
labelmat=flat('Буквы — подписи',(.48,.56,.63,1))

def label(scene,collection,text,x,y,size):
    data=bpy.data.curves.new('Подпись '+text,'FONT')
    data.body=text
    data.size=size
    obj=bpy.data.objects.new('Подпись '+text,data)
    collection.objects.link(obj)
    data.materials.append(labelmat)
    obj.location=(x,y,1)
    return obj

label(library,support,'A — Z',-15,10.25,.6)
label(library,support,'a — z',-15,-.35,.6)
materials={}
for filename in sorted({g['file'] for g in layout['glyphs']}):
    image=bpy.data.images.load(OUT+'/assets/'+filename,check_existing=False)
    image.pack()
    image.filepath='//assets/'+filename
    mat=bpy.data.materials.new('Буквы — неизменный атлас '+filename)
    mat.use_nodes=True
    mat.surface_render_method=next(e.identifier for e in mat.bl_rna.properties['surface_render_method'].enum_items if e.identifier=='BLENDED')
    mat.use_transparent_shadow=False
    nodes=mat.node_tree.nodes
    nodes.clear()
    tex=nodes.new('ShaderNodeTexImage')
    tex.image=image
    tex.extension=next(e.identifier for e in tex.bl_rna.properties['extension'].enum_items if e.identifier=='CLIP')
    tex.interpolation=next(e.identifier for e in tex.bl_rna.properties['interpolation'].enum_items if e.identifier=='Linear')
    em=nodes.new('ShaderNodeEmission')
    transparent=nodes.new('ShaderNodeBsdfTransparent')
    mix=nodes.new('ShaderNodeMixShader')
    output=nodes.new('ShaderNodeOutputMaterial')
    links=mat.node_tree.links
    links.new(tex.outputs['Color'],em.inputs[0])
    links.new(tex.outputs['Alpha'],mix.inputs[0])
    links.new(transparent.outputs[0],mix.inputs[1])
    links.new(em.outputs[0],mix.inputs[2])
    links.new(mix.outputs[0],output.inputs['Surface'])
    materials[filename]=mat

pixel_base=2/layout['normalization']['capital_reference_px']
glyphmap={}
export_metrics=[]
indices={'upper':0,'lower':0}
for info in layout['glyphs']:
    char=info['char']
    case=info['case']
    scale=pixel_base*info['render_scale']
    x0,y0,x1,y1=info['paddedrect']
    bx0,by0,bx1,by1=info['bbox_global']
    baseline=info['baseline_global_y']
    l,r=(x0-bx0)*scale,(x1-bx0)*scale
    b,t=(baseline-y1)*scale,(baseline-y0)*scale
    obj=quad('Latin '+info['id'],[(l,b,0),(r,b,0),(r,t,0),(l,t,0)],upper if case=='upper' else lower,materials[info['file']])
    uv=obj.data.uv_layers.new(name='Только собственная буква')
    w,h=info['size']
    coords=[(x0/w,1-y1/h),(x1/w,1-y1/h),(x1/w,1-y0/h),(x0/w,1-y0/h)]
    for item,value in zip(uv.data,coords):
        item.uv=value
    advance=(bx1-bx0)*scale+.15
    obj['glyph_char']=char
    obj['glyph_case']='symbol' if char=='_' else case
    obj['glyph_id']=info['id']
    obj['glyph_advance']=advance
    obj['baseline_origin']='Left ink edge at x0, shared row baseline at y0; descenders extend below.'
    obj['atlas_file']=info['file']
    obj['atlas_rect']=[x0,y0,x1,y1]
    obj['pixel_scale']=scale
    obj['baseline_px']=baseline
    obj['glyph_bounds_local']=[l,b,r,t]
    obj.asset_mark()
    obj.asset_data.description='BRAT_OK worn ivory metal glyph '+char+'; transparent painted sprite; baseline origin and glyph_advance supplied.'
    for tag in ['BRAT_OK','Latin',obj['glyph_case']]:
        obj.asset_data.tags.new(tag)
    index=indices[case]
    indices[case]+=1
    obj.location=(-15+(index%9)*3.4,(7.6 if case=='upper' else -3.0)-(index//9)*3.0,0)
    glyphmap[char]=obj
    export_metrics.append({'char':char,'case':obj['glyph_case'],'object':obj.name,'file':info['id']+'.png','advance':advance,'bounds':[l,b,r,t],'baseline':0,'pixel_scale':scale})

def compose_word(text,baseline,scale):
    root=bpy.data.objects.new('Надпись '+text,None)
    examplescol.objects.link(root)
    root['text']=text
    root['assembly']='Independent instances of marked Latin glyph assets; shared immutable artwork.'
    cursor=0
    for index,char in enumerate(text):
        if char==' ':
            cursor+=.9*scale
            continue
        sourceletter=glyphmap[char]
        obj=bpy.data.objects.new('Надпись '+text+' — '+str(index)+' '+char,sourceletter.data)
        examplescol.objects.link(obj)
        obj.parent=root
        obj.location=(cursor,0,0)
        obj.scale=(scale,scale,scale)
        cursor+=sourceletter['glyph_advance']*scale
    root.location=(-cursor/2,baseline,0)
    return root

compose_word('BRAT_OK',3.2,.72)
compose_word('StarCraft',-.4,.72)
