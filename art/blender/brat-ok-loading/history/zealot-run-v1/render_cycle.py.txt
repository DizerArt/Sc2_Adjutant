import bpy, json

NAME = "BRAT_OK — зилот: бег на носках"
EXPECTED = "<ORIGINAL_PROJECT>/output/brat-ok-loading/zealot-run-v1/BRAT_OK_zealot_run_v1.blend"
WORK = "<ORIGINAL_PROJECT>/work/brat-ok-zealot-run-v1"
start = FRAME_START
end = FRAME_END
if bpy.context.scene.name != NAME or bpy.data.filepath.replace("\\", "/") != EXPECTED:
    raise RuntimeError("File or active scene changed; inspect before rendering.")
if not (1 <= start <= end <= 49 and end-start < 8):
    raise RuntimeError("Render batches must have at most eight frames in 1..49.")
s = bpy.context.scene
saved_frame = s.frame_current
saved_percent = s.render.resolution_percentage
saved_path = s.render.filepath
s.render.resolution_percentage = 50
states = []
try:
    for frame in range(start, end+1):
        s.frame_set(frame)
        visible = [o["phase_index"] for o in s.objects if o.get("phase_index") is not None and not o.hide_render]
        if len(visible) != 1:
            raise RuntimeError("Each frame must expose exactly one whole-body pose.")
        states.append({"frame": frame, "pose": visible[0]})
        s.render.filepath = WORK + ("/loop-end.png" if frame == 49 else "/frames/frame-" + str(frame-1).zfill(2) + ".png")
        bpy.ops.render.render(write_still=True)
finally:
    s.frame_set(saved_frame)
    s.render.resolution_percentage = saved_percent
    s.render.filepath = saved_path
print(json.dumps({"rendered": len(states), "states": states, "restored_frame": saved_frame}))
