    if spec[low] >= spec[high]:
        raise RuntimeError("Invalid smooth mask boundaries.")

    raise RuntimeError("Motion amplitudes must be nonnegative.")

    raise RuntimeError("Invalid claw exclusion or body window.")

def smooth(value):
    t = min(1.0, max(0.0, value))
    return t*t*(3-2*t)

def offsets(x, y):
    progress = min(1.0,max(0.0,(x-spec["tail_x_start"])/(spec["tail_x_full"]-spec["tail_x_start"])))
    release = smooth((x-spec["tail_y_release_start"])/(spec["tail_y_release_full"]-spec["tail_y_release_start"]))
    below = smooth((y-spec["tail_y_start"])/(spec["tail_y_full"]-spec["tail_y_start"]))
    weight = smooth(progress)*(below*(1-release)+release)
    cx,cy = spec["claw_center"]
    rx,ry = spec["claw_radii"]
    claw_distance = math.sqrt(((x-cx)/rx)**2+((y-cy)/ry)**2)
    weight *= smooth((claw_distance-1)/spec["claw_fade_ratio"])
    belly_pin = (1-smooth((x-spec["belly_x_pin_end"])/(spec["belly_x_release_end"]-spec["belly_x_pin_end"])))*smooth((y-spec["belly_y_start"])/(spec["belly_y_full"]-spec["belly_y_start"]))
    weight *= 1-belly_pin
    phase = 2*math.pi*spec["tail_waves"]*progress
    # The lowermost row cannot move through the fixed ground contact.
    amp_y = min(spec["tail_amp_y"], max(0.0,contact-y-2))*weight
    amp_x = spec["tail_amp_x"]*weight
    body_y = smooth((spec["body_y_pin"]-y)/(spec["body_y_pin"]-spec["body_y_full"]))
    lx0,lx1,rx0,rx1 = spec["body_x_window"]
    body_x = smooth((x-lx0)/(lx1-lx0))*smooth((rx1-x)/(rx1-rx0))
    body = body_y*body_x
    return [(amp_x*math.cos(phase),amp_y*math.sin(phase)),(-amp_x*math.sin(phase),amp_y*math.cos(phase)),(spec["body_amp_x"]*body,0),(0,spec["body_amp_y"]*body)]

def flat(name,color):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    em = nodes.new("ShaderNodeEmission")
    em.inputs["Color"].default_value = color
    out = nodes.new("ShaderNodeOutputMaterial")
    mat.node_tree.links.new(em.outputs[0],out.inputs["Surface"])
    return mat

def quad(name,coords,material):
    mesh = bpy.data.meshes.new(name+" — плоскость")
    mesh.from_pydata(coords,[],[(0,1,2,3)])
    obj = bpy.data.objects.new(name,mesh)
    s.collection.objects.link(obj)
    mesh.materials.append(material)
    return obj
