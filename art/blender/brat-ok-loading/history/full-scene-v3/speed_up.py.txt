import bpy, json
s = bpy.context.scene
expected = '<ORIGINAL_PROJECT>/output/brat-ok-loading/full-scene-v2/BRAT_OK_full_scene_v2.blend'
if s.name != 'BRAT_OK — хоровод: четверка и ультралиск' or bpy.data.filepath.replace('\\','/') != expected:
    raise RuntimeError('Expected the current v2 queue project.')
if s.get('biped_speed_multiplier'):
    raise RuntimeError('Speed adjustment already applied.')
if bpy.context.screen and bpy.context.screen.is_animation_playing:
    bpy.ops.screen.animation_cancel(restore_frame=False)
bpy.ops.wm.save_as_mainfile(filepath='<ORIGINAL_PROJECT>/work/brat-ok-full-scene-v3/before-speed.blend',copy=True,relative_remap=False)
routes = json.loads(s['routes_json'])
source_counts = {sc.name:len(sc.objects) for sc in bpy.data.scenes}
source_frames = {sc.name:sc.frame_current for sc in bpy.data.scenes if sc!=s}

def curves(action):
    return [fc for layer in action.layers for strip in layer.strips for bag in strip.channelbags for fc in bag.fcurves]

def signature(action):
    return [(fc.data_path,fc.array_index,[(tuple(k.co),k.interpolation) for k in fc.keyframe_points]) for fc in curves(action)]

root_signatures = {r['root']:signature(s.objects[r['root']].animation_data.action) for r in routes}
targets = [r for r in routes if r['unit'] in ['Мародёр','Зилот']]
if any(r['cycle']!=48 for r in targets) or len(targets)!=2:
    raise RuntimeError('Both biped cycles must initially have 48 frames.')
target_objects = [o for r in targets for o in s.objects[r['root']].children]
other_actions = {o.name:o.animation_data.action for sc in bpy.data.scenes for o in sc.objects if o not in target_objects and o.animation_data and o.animation_data.action}
for route in targets:
    route['cycle'] = 32
    route['offset'] = route['offset'] * 2 // 3
    root = s.objects[route['root']]
    root['native_cycle_frames'] = 32
    root['cycle_phase_offset'] = route['offset']
    root['gait_speed_multiplier'] = 1.5
    if len(root.children)!=8:
        raise RuntimeError('Expected eight whole-body poses.')
    for obj in root.children:
        idx = int(obj['phase_index'])
        obj.animation_data_clear()
        obj['hold_frames'] = 4
        for f in range(1,578):
            phase = ((f-1+route['offset']) % 32) // 4
            obj.hide_render = phase != idx
            obj.hide_viewport = phase != idx
            obj.keyframe_insert(data_path='hide_render',frame=f)
            obj.keyframe_insert(data_path='hide_viewport',frame=f)
        for fc in curves(obj.animation_data.action):
            for key in fc.keyframe_points:
                key.interpolation = next(e.identifier for e in key.bl_rna.properties['interpolation'].enum_items if e.identifier=='CONSTANT')
s['routes_json'] = json.dumps(routes,ensure_ascii=False)
s['biped_speed_multiplier'] = 1.5
s['biped_cycle_seconds'] = 32/30
checks = {
    'root_traversal_actions_exactly_unchanged':all(signature(s.objects[name].animation_data.action)==sig for name,sig in root_signatures.items()),
    'all_other_object_action_ids_unchanged':all(bpy.data.objects[name].animation_data.action==action for name,action in other_actions.items()),
    'all_scene_counts_preserved':all(len(bpy.data.scenes[name].objects)==count for name,count in source_counts.items()),
    'other_scene_frames_preserved':all(bpy.data.scenes[name].frame_current==f for name,f in source_frames.items()),
    'exact_speed_ratio_1_5':48/32==1.5,
}
sequence_ok = True
phase_sets = {r['unit']:set() for r in targets}
for f in range(1,578):
    s.frame_set(f)
    for route in targets:
        shown = [o for o in s.objects[route['root']].children if not o.hide_render]
        expected_phase = ((f-1+route['offset']) % 32)//4
        sequence_ok = sequence_ok and len(shown)==1 and int(shown[0]['phase_index'])==expected_phase and not shown[0].hide_viewport
        if shown:
            phase_sets[route['unit']].add(int(shown[0]['phase_index']))
checks['exact_visibility_sequence_all577frames'] = sequence_ok
checks['all_eight_poses_preserved_both_units'] = all(values==set(range(8)) for values in phase_sets.values())
checks['complete_cycle_32_frames_four_frames_each'] = all(o.get('hold_frames')==4 for o in target_objects) and 576%32==0
checks['far_offscreen_after_frame288'] = all(max((s.objects[r['root']].location.x+v.co.x*s.objects[r['root']].scale.x) for child in s.objects[r['root']].children for v in child.data.vertices)<-9.6 for r in targets)
# The above sample is frame577, where roots reset. Compute frame289 explicitly.
s.frame_set(289)
checks['far_offscreen_after_frame288'] = all(max((s.objects[r['root']].location.x+v.co.x*s.objects[r['root']].scale.x) for child in s.objects[r['root']].children for v in child.data.vertices)<-9.6 for r in targets)
if not all(checks.values()):
    raise RuntimeError(json.dumps(checks))
s.frame_set(180)
s.render.filepath='<ORIGINAL_PROJECT>/output/brat-ok-loading/full-scene-v3/preview.png'
bpy.ops.wm.save_as_mainfile(filepath='<ORIGINAL_PROJECT>/output/brat-ok-loading/full-scene-v3/BRAT_OK_full_scene_v3.blend',relative_remap=False)
print(json.dumps({'all_passed':all(checks.values()),'checks':checks,'units':[{'unit':r['unit'],'old_cycle':48,'new_cycle':32,'seconds':32/30,'hold_frames':4,'offset':r['offset']} for r in targets],'global_loop_seconds':19.2,'translation_speed':3.6,'source_counts':source_counts},ensure_ascii=False))
