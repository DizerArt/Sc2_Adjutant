def material(filename):
    img = bpy.data.images.load(ROOT + "/assets/" + filename, check_existing=False)
    img.pack()
    img.filepath = "//assets/" + filename
    mat = bpy.data.materials.new("Новый рисунок — " + filename)
    mat.use_nodes = True
    methods = [i.identifier for i in mat.bl_rna.properties["surface_render_method"].enum_items]
    if "BLENDED" in methods:
        mat.surface_render_method = methods[methods.index("BLENDED")]
    mat.use_transparent_shadow = False
    nodes = mat.node_tree.nodes
    nodes.clear()
    tex = nodes.new("ShaderNodeTexImage")
    tex.image = img
    extensions = [i.identifier for i in tex.bl_rna.properties["extension"].enum_items]
    tex.extension = extensions[extensions.index("CLIP")]
    emission = nodes.new("ShaderNodeEmission")
    transparent = nodes.new("ShaderNodeBsdfTransparent")
    mix = nodes.new("ShaderNodeMixShader")
    output = nodes.new("ShaderNodeOutputMaterial")
    links = mat.node_tree.links
    links.new(tex.outputs["Color"], emission.inputs["Color"])
    links.new(tex.outputs["Alpha"], mix.inputs[0])
    links.new(transparent.outputs[0], mix.inputs[1])
    links.new(emission.outputs[0], mix.inputs[2])
    links.new(mix.outputs[0], output.inputs["Surface"])
    return mat

def plane(name, x0, y0, width, height, z, mat, col):
    mesh = bpy.data.meshes.new(name + " — полная плоскость")
    mesh.from_pydata([(0,0,0),(width,0,0),(width,height,0),(0,height,0)], [], [(0,1,2,3)])
    uv = mesh.uv_layers.new(name="Полный рисунок UV")
    for item, coord in zip(uv.data, [(0,0),(1,0),(1,1),(0,1)]):
        item.uv = coord
    obj = bpy.data.objects.new(name, mesh)
    col.objects.link(obj)
    obj.location = (x0,y0,z)
    obj.data.materials.append(mat)
    return obj
