def flat_material(name,color):
    m=bpy.data.materials.new(name)
    m.use_nodes=True
    ns=m.node_tree.nodes
    ns.clear()
    e=ns.new("ShaderNodeEmission")
    e.inputs["Color"].default_value=color
    out=ns.new("ShaderNodeOutputMaterial")
    m.node_tree.links.new(e.outputs[0],out.inputs["Surface"])
    return m
def quad(name,coords,material,col):
    mesh=bpy.data.meshes.new(name+" — плоскость")
    mesh.from_pydata(coords,[],[(0,1,2,3)])
    obj=bpy.data.objects.new(name,mesh)
    col.objects.link(obj)
    obj.data.materials.append(material)
    return obj
