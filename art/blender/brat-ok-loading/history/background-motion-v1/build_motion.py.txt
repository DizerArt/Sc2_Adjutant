def link_or_value(value,socket,tree_links):
    if isinstance(value,(int,float)):
        socket.default_value=value
    else:
        tree_links.new(value,socket)
def calc(tree,op,a,b=None):
    n=tree.nodes.new("ShaderNodeMath")
    n.operation=ops[ops.index(op)]
    link_or_value(a,n.inputs[0],tree.links)
    if b is not None: link_or_value(b,n.inputs[1],tree.links)
    return n.outputs[0]
def anim(tree,name,values):
    n=tree.nodes.new("ShaderNodeValue")
    n.name=name
    n.label=name
    for i,value in enumerate(values):
        n.outputs[0].default_value=value
        n.outputs[0].keyframe_insert(data_path="default_value",frame=i+1)
    controls.append((tree,n))
    return n.outputs[0]
def wave(tree,name,cycles,phase=0):
    return anim(tree,name,[math.sin(2*math.pi*cycles*i/384+phase) for i in range(385)])
def ellipse(tree,u,v,cx,cy,rx,ry):
    x=calc(tree,"DIVIDE",calc(tree,"SUBTRACT",u,cx),rx)
    y=calc(tree,"DIVIDE",calc(tree,"SUBTRACT",v,cy),ry)
    d=calc(tree,"ADD",calc(tree,"MULTIPLY",x,x),calc(tree,"MULTIPLY",y,y))
    return calc(tree,"POWER",calc(tree,"MAXIMUM",calc(tree,"SUBTRACT",1,d),0),2)
def add(tree,a,b): return calc(tree,"ADD",a,b)
def mul(tree,a,b): return calc(tree,"MULTIPLY",a,b)
def total(tree,values):
    result=values[0]
    for value in values[1:]: result=add(tree,result,value)
    return result

def effect_material(name,color,strength):
    m=bpy.data.materials.new(name)
    m.use_nodes=True
    methods=[i.identifier for i in m.bl_rna.properties["surface_render_method"].enum_items]
    if "BLENDED" in methods: m.surface_render_method=methods[methods.index("BLENDED")]
    m.use_transparent_shadow=False
    ns=m.node_tree.nodes
    ns.clear()
    e=ns.new("ShaderNodeEmission")
    e.inputs["Color"].default_value=color
    e.inputs["Strength"].default_value=strength
    t=ns.new("ShaderNodeBsdfTransparent")
    mix=ns.new("ShaderNodeMixShader")
    out=ns.new("ShaderNodeOutputMaterial")
    m.node_tree.links.new(t.outputs[0],mix.inputs[1])
    m.node_tree.links.new(e.outputs[0],mix.inputs[2])
    m.node_tree.links.new(mix.outputs[0],out.inputs["Surface"])
    return m,mix
